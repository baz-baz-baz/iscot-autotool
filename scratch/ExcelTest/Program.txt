using System;
using System.IO;
using System.Linq;
using ClosedXML.Excel;

class Program
{
    static void Main(string[] args)
    {
        string logPath = Path.Combine(Environment.GetFolderPath(Environment.SpecialFolder.Desktop), "LOG & DUMP");
        var files = Directory.GetFiles(logPath, "Report Interventi*.xls*", SearchOption.AllDirectories);
        if (files.Length == 0) return;
        string file = files[0];
        try {
            using (var wb = new XLWorkbook(file)) {
                var ws = wb.Worksheets.FirstOrDefault();
                foreach(var dv in ws.DataValidations) {
                    if (dv.Ranges.Any(r => r.FirstAddress.ColumnNumber == 27 || r.LastAddress.ColumnNumber == 27)) {
                        Console.WriteLine("DV Range: " + string.Join(",", dv.Ranges.Select(r => r.RangeAddress.ToString())) + " -> " + dv.Value);
                    }
                }
            }
        } catch (Exception ex) { Console.WriteLine(ex.Message); }
    }
}
